払ワイパーで払われるNEWWIPER-SWEEP

下の中心を軸にワイパーが振れて、通った所の粒が横へ払われていきます。

コードを見る
// 画面の下の真ん中を軸にした、ワイパーの動き。通った所の粒が、振れる向きへ払われる
vec3 userMethod(vec2 p, float amount) {
  vec2 c = partC(), l = p - c;
  vec2 u = (c - vec2(0., -.5 * uSize.y)) / uSize.y;           // 軸（下の真ん中）から見た位置
  float turn = atan(u.x, u.y) / 3.1416 + .5;                  // 左 0 から 右 1 へ、ワイパーが振れる
  float delay = turn * .45 + ph(1.) * .05;
  float e = smoothstep(delay, delay + .22, amount);
  vec2 push = normalize(vec2(u.y, -u.x) + vec2(.0001)) * e * e * (.45 + ph(2.) * .6) * uSize.y;
  l = rot((ph(3.) - .5) * 7. * e) * l * (1. - .3 * e);
  gKind = 2.;                                                 // 粒の見た目: 2 = 丸い点
  gAlpha = 1. - .3 * e;
  gLocal = 1.;
  return vec3(c + l + push, (ph(4.) - .5) * .3 * e);
}

跳ぴょんぴょん跳ねるタイルNEWHOP-WAVE

斜めに波が走って、タイルが順にぴょんぴょんと跳ね上がって戻ります。

// 左下から右上へ斜めに波が走り、タイルが順に跳ねて、少しずつ小さくなる跳ね方で落ち着く。強さ(amount)が戻ると元へ帰る
const {createImageTiles,size,smooth}=api;
const tiles=createImageTiles();

return {update(frame){
 const t=Math.min(1,frame.amount/.8);
 for(const piece of tiles){
  const x=piece.home.x,y=piece.home.y,diag=((x/half)+(y/half)+2)/4;      // 0 = 左下、1 = 右上
  const local=Math.max(0,Math.min(1,t*1.1-diag*.35));                    // タイルごとの進み具合
  const hop=Math.abs(Math.sin(local*Math.PI*2.5))*(1-.5*local)*half*.22;   // 2回半、だんだん低く（強さが最大のときも跳ねている）
  const grow=smooth(Math.min(1,local*3))*(1-smooth(Math.max(0,local*2-1)));

  piece.mesh.rotation.set(0,0,Math.sin(local*Math.PI*2.5)*.25*(1-.5*local));
  piece.mesh.scale.setScalar(1+.15*grow);
 }

市市松が上下にずれるNEWCHECKER-SHIFT

市松模様のマスが縮みながら、交互に上と下へずれて崩れていきます。

コードを見る
// 絵を12×12のマスに分け、市松模様の片方は上へ、もう片方は下へ、縮みながらずらす
vec3 userMethod(vec2 p, float amount) {
  vec2 c = partC();
  vec2 id = floor((c / uSize + .5) * 12.);                    // マスの番号
  float odd = mod(id.x + id.y, 2.);                           // 市松模様の 0 / 1
  vec2 mid = ((id + .5) / 12. - .5) * uSize;                  // マスの真ん中
  float delay = hash(id) * .25;
  float e = smoothstep(delay, delay + .3, amount);
  vec2 q = mid + (p - mid) * (1. - .55 * e);                  // マスごとに縮む
  q.y += (odd * 2. - 1.) * e * uSize.y * .45;                 // 片方は上、もう片方は下
  gKind = 2.;                                                 // 粒の見た目: 2 = 丸い点
  gAlpha = 1. - .25 * e;
  gLocal = 1.;
  return vec3(q, 0.);
}
